async function fetchServerStats() { const statusEl = document.getElementById('server-status'); const playersEl = document.getElementById('online-players'); try { // FalixNodes IP ve portumuz üzerinden anlık durum sorgulama const res = await fetch('https://api.mcsrvstat.us/2/eu9-free.falixserver.net:24982'); const data = await res.json(); if (data.online) { statusEl.innerText = 'Çevrimiçi'; statusEl.style.color = '#55ff55'; playersEl.innerText = `${data.players.online} / ${data.players.max}`; } else { statusEl.innerText = 'Kapalı'; statusEl.style.color = '#ff5555'; playersEl.innerText = '0 / 0'; } } catch (e) { statusEl.innerText = 'Kapalı'; statusEl.style.color = '#ff5555'; playersEl.innerText = '0 / 0'; } } // Sayfa açıldığında çalıştır ve her 30 saniyede bir otomatik yenile fetchServerStats(); setInterval(fetchServerStats, 30000);
